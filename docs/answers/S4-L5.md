# S4-L5 · 모범 풀이

```python
def solve_case():
    n = int(input())
    g = [list(map(int, input().split())) for _ in range(n)]
    for r in range(n):
        for c in range(n):
            if r == 0 and c == 0:
                continue
            best = 10 ** 9
            if r:
                best = min(best, g[r - 1][c])
            if c:
                best = min(best, g[r][c - 1])
            g[r][c] += best
    print(g[-1][-1])
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
