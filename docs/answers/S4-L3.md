# S4-L3 · 모범 풀이

```python
def solve_case():
    n = int(input())
    g = [list(input().strip()) for _ in range(n)]
    answer = 0
    for r in range(n):
        for c in range(n):
            if g[r][c] != '1':
                continue
            answer += 1
            g[r][c] = '0'
            st = [(r, c)]
            while st:
                x, y = st.pop()
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = (x + dx, y + dy)
                    if 0 <= nx < n and 0 <= ny < n and (g[nx][ny] == '1'):
                        g[nx][ny] = '0'
                        st.append((nx, ny))
    print(answer)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
