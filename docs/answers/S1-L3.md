# S1-L3 · 모범 풀이

```python
def solve_case():
    from collections import deque
    r, c = map(int, input().split())
    g = [input().strip() for _ in range(r)]
    sr, sc = next(((i, j) for i in range(r) for j in range(c) if g[i][j] == 'S'))
    q = deque([(sr, sc, 0)])
    seen = {(sr, sc)}
    ans = -1
    while q:
        i, j, d = q.popleft()
        if g[i][j] == 'E':
            ans = d
            break
        for di, dj in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            ni, nj = (i + di, j + dj)
            if 0 <= ni < r and 0 <= nj < c and (g[ni][nj] != '#') and ((ni, nj) not in seen):
                seen.add((ni, nj))
                q.append((ni, nj, d + 1))
    print(ans)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
