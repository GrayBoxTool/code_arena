# S1-L5 · 모범 풀이

```python
def solve_case():
    n, k, target = map(int, input().split())
    a = list(map(int, input().split()))
    answer = 0
    for mask in range(1 << n):
        count = 0
        total = 0
        for i in range(n):
            if mask & 1 << i:
                count += 1
                total += a[i]
        if count == k and total == target:
            answer += 1
    print(answer)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
