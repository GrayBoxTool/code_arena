# S1-L5 · 모범 풀이

```python
def solve_case():
    n,m=map(int,input().split())
    g=[input().strip() for _ in range(n)]
    p=[[0] for _ in range(3)]
    for row in g:
        for i,c in enumerate("WBR"): p[i].append(p[i][-1]+m-row.count(c))
    print(min(p[0][a]+p[1][b]-p[1][a]+p[2][n]-p[2][b] for a in range(1,n-1) for b in range(a+1,n)))
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
