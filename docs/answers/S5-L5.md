# S5-L5 · 모범 풀이

```python
def solve_case():
    n, v = map(int, input().split())
    a = list(map(int, input().split()))
    ans = -1
    for i in range(n):
        for j in range(i + 1, n):
            for k in range(j + 1, n):
                s = a[i] + a[j] + a[k]
                if s <= v:
                    ans = max(ans, s)
    print(ans)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
