# S3-L3 · 모범 풀이

```python
def solve_case():
    n, t = map(int, input().split())
    a = sorted(map(int, input().split()))
    l = 0
    r = n - 1
    ans = 0
    while l < r:
        s = a[l] + a[r]
        if s == t:
            ans += 1
            l += 1
            r -= 1
        elif s < t:
            l += 1
        else:
            r -= 1
    print(ans)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
