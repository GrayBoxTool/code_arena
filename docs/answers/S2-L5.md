# S2-L5 · 모범 풀이

```python
def solve_case():
    n,m,k=map(int,input().split())
    a=list(map(int,input().split()))
    p=0
    for _ in range(k):
        p=(p+m-1)%len(a)+1
        a.insert(p,a[p-1]+a[p%len(a)])
    print(*a[-10:][::-1])
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
