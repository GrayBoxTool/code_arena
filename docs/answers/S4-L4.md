# S4-L4 · 모범 풀이

```python
def solve_case():
    n = int(input())
    a = list(map(int, input().split()))
    h = [0]
    for x in a:
        h.append(x)
        i = len(h) - 1
        while i > 1 and h[i] < h[i // 2]:
            h[i], h[i // 2] = (h[i // 2], h[i])
            i //= 2
    answer = 0
    i = n // 2
    while i:
        answer += h[i]
        i //= 2
    print(answer)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
