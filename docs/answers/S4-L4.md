# S4-L4 · 모범 풀이

```python
def solve_case():
    p,q=map(int,input().split())
    def coord(v):
        d=1
        while d*(d+1)//2<v: d+=1
        x=v-d*(d-1)//2
        return x,d+1-x
    x,y=coord(p)
    u,v=coord(q)
    x+=u; y+=v
    print((x+y-2)*(x+y-1)//2+x)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
