# S2-L3 · 모범 풀이

```python
def solve_case():
    n, m = map(int, input().split())
    g = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        g[u].append(v)
        g[v].append(u)
    seen = set()
    ans = 0
    for i in range(1, n + 1):
        if i in seen:
            continue
        ans += 1
        seen.add(i)
        stack = [i]
        while stack:
            for v in g[stack.pop()]:
                if v not in seen:
                    seen.add(v)
                    stack.append(v)
    print(ans)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
