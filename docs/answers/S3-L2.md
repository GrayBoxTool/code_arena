# S3-L2 · 모범 풀이

```python
def solve_case():
    from collections import deque
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    q = deque(enumerate(a))
    cnt = 0
    while q:
        i, p = q.popleft()
        if any((other > p for _, other in q)):
            q.append((i, p))
        else:
            cnt += 1
            if i == k:
                print(cnt)
                break
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
