# S2-L4 · 모범 풀이

```python
def solve_case():
    n, v = map(int, input().split())
    coins = list(map(int, input().split()))
    dp = [10 ** 9] * (v + 1)
    dp[0] = 0
    for x in range(1, v + 1):
        for c in coins:
            if c <= x:
                dp[x] = min(dp[x], dp[x - c] + 1)
    print(dp[v] if dp[v] < 10 ** 9 else -1)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
