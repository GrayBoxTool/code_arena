# S3-L4 · 모범 풀이

```python
def solve_case():
    import heapq
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    heapq.heapify(a)
    total = 0
    for _ in range(k):
        x = heapq.heappop(a) + heapq.heappop(a)
        total += x
        heapq.heappush(a, x)
    print(total)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
