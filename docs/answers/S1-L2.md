# S1-L2 · 모범 풀이

```python
def solve_case():
    s = input().strip()
    st = []
    pair = {')': '(', ']': '['}
    ok = True
    for c in s:
        if c in '([':
            st.append(c)
        elif not st or st.pop() != pair[c]:
            ok = False
            break
    print('YES' if ok and (not st) else 'NO')
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
