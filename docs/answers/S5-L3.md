# S5-L3 · 모범 풀이

```python
def solve_case():
    g = [input().split() for _ in range(3)]
    found = set()

    def dfs(r, c, s):
        if len(s) == 4:
            found.add(s)
            return
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = (r + dr, c + dc)
            if 0 <= nr < 3 and 0 <= nc < 3:
                dfs(nr, nc, s + g[nr][nc])
    for r in range(3):
        for c in range(3):
            dfs(r, c, g[r][c])
    print(len(found))
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
