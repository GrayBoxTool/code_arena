# S5-L4 · 모범 풀이

```python
def solve_case():
    p, a, b = map(int, input().split())

    def search(target):
        l, r = (1, p)
        count = 0
        while l <= r:
            count += 1
            m = (l + r) // 2
            if m == target:
                return count
            if target < m:
                r = m - 1
            else:
                l = m + 1
    x, y = (search(a), search(b))
    print('A' if x < y else 'B' if y < x else 'TIE')
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
