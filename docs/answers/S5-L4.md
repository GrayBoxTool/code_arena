# S5-L4 · 모범 풀이

```python
def solve_case():
    n,m=map(int,input().split())
    a=list(map(int,input().split()))
    b=list(map(int,input().split()))
    f=[0]*n
    for budget in b:
        for i,cost in enumerate(a):
            if cost<=budget:
                f[i]+=1
                break
    print(f.index(max(f))+1)
T = int(input())
for tc in range(1, T + 1):
    print(f'#{tc} ', end='')
    solve_case()
```
